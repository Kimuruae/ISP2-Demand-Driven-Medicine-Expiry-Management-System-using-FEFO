    @staticmethod
    def _validate_positive_integer(value, field_name): #Validate a strictly positive whole-number input.
        if isinstance(value, bool):
            raise ValueError(f"{field_name} must be a positive whole number.")

        try:
            number = int(value)
        except (TypeError, ValueError):
            raise ValueError(f"{field_name} must be a positive whole number.")

        if str(value).strip() != str(number) or number <= 0:
            raise ValueError(f"{field_name} must be a positive whole number.")
        return number

    @staticmethod
    def _get_eligible_batches(medicine, required_valid_until, lock=False):
    #Return batches that satisfy general stock eligibility and prescription-period validity, ordered using FEFO.
        batches = MedicineBatch.objects.filter(
            medicine=medicine,
            medicine__is_active=True,
            quantity_available__gt=0,
            expiry_date__gt=timezone.localdate(),
            expiry_date__gte=required_valid_until,
            status="Active",
        ).order_by("expiry_date", "id")

        if lock:  batches = batches.select_for_update()
        return batches

    @staticmethod
    def recommend_batch(medicine_id, prescription_duration_days):
        duration = FEFOEngine._validate_positive_integer(prescription_duration_days,"Prescription duration",)

        try:
            medicine = Medicine.objects.get(pk=medicine_id,is_active=True,)
        except (Medicine.DoesNotExist, TypeError, ValueError): raise ValueError("The medicine does not exist or is inactive.")

        today = timezone.localdate()
        required_valid_until = today + timedelta(days=duration - 1)
        return FEFOEngine._get_eligible_batches(medicine,required_valid_until,).first()

        """
        Dispensing medicine using prescription-period-aware FEFO. This is mainly for patient's safety
        Automatic allocation uses the earliest-expiring eligible batches.
        Manual selection is permitted only when the selected batch meets
        the same prescription-period validity requirements.
        """
        quantity = FEFOEngine._validate_positive_integer(total_qty_to_dispense, "Dispensing quantity",)
        duration = FEFOEngine._validate_positive_integer(prescription_duration_days,"Prescription duration",)

        try:
            medicine = Medicine.objects.get( pk=medicine_id,is_active=True,)
        except (Medicine.DoesNotExist, TypeError, ValueError):
            raise ValueError("The medicine does not exist or is inactive.")
        today = timezone.localdate()

        # The dispensing date is counted as day one.
        required_valid_until = today + timedelta(days=duration - 1)

        # Lock eligible batches to reduce concurrent stock deduction.
        eligible_batches = list(FEFOEngine._get_eligible_batches(medicine,required_valid_until, lock=True,))

        if not eligible_batches:
            generally_eligible = MedicineBatch.objects.filter(
                medicine=medicine,
                quantity_available__gt=0,
                expiry_date__gt=today,
                status="Active",
            ).exists()

            if generally_eligible:
                raise ValueError(
                    "Stock exists, but no available batch meets the "
                    "required prescription-period validity date "
                    f"({required_valid_until}). Notify the Store Manager."
                )

            expired_exists = MedicineBatch.objects.filter(
                medicine=medicine,
                quantity_available__gt=0,
                expiry_date__lte=today,
            ).exists()

            if expired_exists:
                raise ValueError(
                    "No eligible stock is available. Existing stock is "
                    "expired or otherwise ineligible. Notify the Store Manager."
                )

            raise ValueError(
                "No eligible stock is available. Check depleted stock "
                "or raise a stock-out alert."
            )

        recommended_batch = eligible_batches[0]
        override = False
        batches_to_use = eligible_batches
        reason = (notes or "").strip()

        if selected_batch_id not in (None, ""):
            try:
                selected_id = int(selected_batch_id)
            except (TypeError, ValueError):raise ValueError("The selected batch is invalid.")
            selected_batch = next( (batch for batch in eligible_batches if batch.pk == selected_id),None, )

            if selected_batch is None:
                raise ValueError(
                    "The selected batch is unavailable or does not "
                    "meet the prescription-period validity requirements."

            override = selected_batch.pk != recommended_batch.pk

            if override and not reason:
                raise ValueError("Enter a reason for overriding the FEFO recommendation.")

            batches_to_use = [selected_batch]

    @classmethod
    @transaction.atomic
    def calculate_3_month_forecast(cls, medicine_id):
        """
        Calculate and record a 30-day demand forecast.

        Replenishment logic:
        - Forecast demand from three historical 30-day periods.
        - Include safety-stock coverage in the target.
        - Recommend purchasing only the stock shortfall.
        - Refresh an existing pending recommendation when possible.

        The target uses safety_stock_days as additional days of
        forecast coverage. Supplier lead time is not currently
        modelled separately; add it when a reliable field exists.
        """
        medicine = Medicine.objects.select_for_update().get(id=medicine_id)
        now = timezone.now()
        today = timezone.localdate()

        # Define three consecutive 30-day historical periods.
        period_1_start = now - timedelta(days=30)
        period_1_end = now

        period_2_start = now - timedelta(days=60)
        period_2_end = period_1_start

        period_3_start = now - timedelta(days=90)
        period_3_end = period_2_start

        # Retrieve historical dispensing totals.
        vol_1, count_1 = cls._get_period_consumption(medicine, period_1_start, period_1_end)
        vol_2, count_2 = cls._get_period_consumption(medicine, period_2_start, period_2_end)
        vol_3, count_3 = cls._get_period_consumption(medicine, period_3_start, period_3_end)
        volumes = [vol_1, vol_2, vol_3]
        transaction_count = count_1 + count_2 + count_3

        # Simple moving average: equal weight for each 30-day period.
        moving_average = cls._round_units(Decimal(sum(volumes)) / Decimal(3))

        # A score of zero means no recorded demand or very unstable
        # demand. It should not be interpreted as forecast probability.
        consistency_score = cls._calculate_consistency_score(volumes)

        # Record whether the historical evidence is limited.
        # Three periods with no records are different from three
        # observed periods that happened to have zero consumption.
        insufficient_history = transaction_count == 0

        # Store forecast dates as dates, consistent with the model
        # fields shown in the existing implementation.
        forecast_start = today
        forecast_end = today + timedelta(days=cls.PERIOD_DAYS)

        forecast_entry, _created = DemandForecast.objects.update_or_create(
            medicine=medicine,
            forecast_period_start=forecast_start,
            forecast_period_end=forecast_end,
            defaults={
                "forecasted_demand_quantity": moving_average,
                "moving_avg_3month": moving_average,
                "forecast_method": DemandForecast.ForecastMethod.MOVING_AVERAGE,
                "history_start_date": period_3_start.date(),
                "history_end_date": today,
                "historical_transaction_count": transaction_count,
                "confidence_score": consistency_score,
            },
        )

        # Obtain stock eligible for FEFO dispensing.
        eligible_stock, eligible_batches = (
            cls._get_fefo_eligible_stock(medicine)
        )

        # Estimate stock likely to expire during the forecast horizon.
        expiring_stock = cls._get_expiring_stock(eligible_batches,forecast_end,)

        # Safety stock is additional coverage, expressed in days.
        safety_days = max(0,int(getattr(medicine, "safety_stock_days", 0) or 0),)

        # Target = demand for the next 30 days + safety-stock coverage.
        # This is a basic policy; supplier lead time should be added
        # separately once the model contains a reliable lead-time field.
        target_stock = cls._round_units(Decimal(moving_average)* Decimal(cls.PERIOD_DAYS + safety_days)/ Decimal(cls.PERIOD_DAYS))

        # Replenishment is the shortfall against target inventory.
        recommended_quantity = max(0,target_stock - eligible_stock,)

        # Determine whether the current inventory is below target.
        needs_replenishment = recommended_quantity > 0

        if needs_replenishment:
            # HIGH means the usable stock cannot cover half the
            # forecasted 30-day demand. This is a transparent rule,
            # not a clinical risk classification.
            if ( moving_average > 0 and eligible_stock <= moving_average / 2
            ): urgency = "HIGH"
            else: urgency = "MEDIUM"

            reasons = [
                f"FEFO-eligible stock: {eligible_stock} units.",
                f"Forecast demand for 30 days: {moving_average} units.",
                f"Target stock including safety coverage: {target_stock} units.",
                f"Estimated replenishment shortfall: {recommended_quantity} units.",
            ]

            if insufficient_history:

                    "Warning: no dispensing records were found in "
                    "the 90-day historical window. Validate demand "
                    "before approving procurement."
                )

                    f"Approximately {expiring_stock} eligible units "
                    "have expiry dates within the forecast horizon. "
                    "Review batch-level expiry dates before ordering."
                )

@role_required('Hospital Admin', 'Hospital Administrator')
def hospital_admin_process_po_view(request, po_id, action):
    if request.method != "POST":
        messages.error(request,"Purchase orders must be processed using a POST request.",)
        return redirect("admin_dashboard")

    normalized_action = action.strip().casefold()
    if normalized_action not in ("approve", "reject"):
        messages.error(request, "Invalid purchase-order action.")
        return redirect("admin_dashboard")

    with transaction.atomic():
        po = get_object_or_404(
            PurchaseOrder.objects.select_for_update().select_related(
                "recommendation"
            ),
            id=po_id,
            status=PurchaseOrder.Status.PENDING_ADMIN_REVIEW,
        )

        recommendation = po.recommendation
        if normalized_action == "approve":
            po.status = PurchaseOrder.Status.APPROVED
            success_message = f"Purchase Order REQ #{po.pk} authorized."

            if recommendation:
                recommendation.status = ProcurementRecommendation.Status.ACTIONED
                recommendation.is_actioned = True

        else:
            po.status = PurchaseOrder.Status.REJECTED
            success_message = f"Purchase Order REQ #{po.pk} declined."

            if recommendation:
                recommendation.status = ProcurementRecommendation.Status.REJECTED
                recommendation.is_actioned = False
        po.save(update_fields=["status"])

        if recommendation:
            recommendation.reviewed_by = request.user
            recommendation.save(
                update_fields=[
                    "status",
                    "is_actioned",
                    "reviewed_by",
                    "updated_at",
                ]
            )
